export const figuresFor = (
  kind: EssayKind,
  text: string,
  placement: "before" | "after",
) =>
  essayFigures[kind].filter(
    (figure) =>
      text.includes(figure.anchor) &&
      (figure.placement ?? "after") === placement,
  );
